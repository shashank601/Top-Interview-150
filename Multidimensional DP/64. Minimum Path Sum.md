class Solution {
public:
    vector<vector<int>> dp;
    int g(vector<vector<int>>& grid, int r, int c) {
        int m = grid.size();
        int n = grid[0].size();
        
        if (r == m - 1 && c == n - 1) return grid[r][c];
        if (r >= m || c >= n) {
            return INT_MAX;
        }
        
        if (dp[r][c] != -1) return dp[r][c];

        int bottom = g(grid, r + 1, c);
        int right = g(grid, r, c + 1);

        return dp[r][c] = min(
            bottom, 
            right
        ) + grid[r][c];
    }

    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();

        dp.assign(m, vector<int>(n, -1));
        return g(grid, 0, 0);
    }
};
